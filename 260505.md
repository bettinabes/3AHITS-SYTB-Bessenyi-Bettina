# script

#!/bin/bash

dir=$1

if [ ! -d "$dir" ]
then
        mkdir "$dir"    
        echo "$dir" > "$dir/$dir.txt"
else
        files=$(ls -1 "$dir" | wc -l)   
        echo "Soll das Directory $dir (mit $files Files) gelöscht werden? [j|n]: "
        read answer
        if [ "$answer" = "j" ] 
        then
                rm -r "$dir"
                mkdir "$dir"    
                echo "$dir" > "$dir/$dir.txt"
        elif [ "$answer" = "n" ]
        then
                exit 0
        fi
fi
