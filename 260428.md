#!/bin/bash

day=$(date +%A)

echo "Es ist $day"

test "$day" = "Monday" && echo " GGP SEW SEW SEW ITPMG"
test "$day" = "Tuesday" && echo " SYTB SYTB ITPM E1 ITSE SYTB"
test "$day" = "Wednesday" && echo " D AM NWT NWT RK E1"
test "$day" = "Thursday" && echo " AM SYTE ITPM BSP BSP"
test "$day" = "Friday" && echo " MEDT ITPA RK ITSE MEDT MEDT"

test "$day" != "Monday" -a \
     "$day" != "Tuesday" -a \
     "$day" != "Wednesday" -a \
     "$day" != "Thursday" -a \
     "$day" != "Friday" \
&& echo  "Kein Stundenplan (Wochenende)"
