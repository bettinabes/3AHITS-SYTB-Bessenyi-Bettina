#!/bin/env bash

while ((counter<=10))
do
    echo $counter
    ((counter++))
done

echo "--- for ---"
# Liste = durch white-spaces getrennter string

data="1 2 3 4 5 6 7"
for d in $data

    echo $d

echo "--- Kommandozeilenargumente ---"

for arg in $@

    echo $arg

for value in {20..25}
do 
    echo $value

IFS=",;"     # internal field seperator

    echo $el

do
   sum=$((sum + arg))
done

echo $sum

max=0

for arg in $@
do
     if (( arg > max))
     then
          max=$arg
     fi
