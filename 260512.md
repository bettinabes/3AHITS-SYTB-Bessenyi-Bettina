Vorgang

- putty.zip herunterladen
- SHA-512-Checksum-Datei mit originalen Hashwert herunterladen 
- Hashwert der eigenen Datei mit openssl berechnen
- Hash mit Wert aus der Checksum-Datei vergleichen
- Wenn beide übereinstimmen ist die Datei nicht verändert worden

#script

```
TMPDIR=$(mktemp -d)

curl -L -o "$TMPDIR/sha512sums" \
https://the.earth.li/~sgtatham/putty/latest/sha512sums

grep "w64/putty.zip$" "$TMPDIR/sha512sums" > "$TMPDIR/line"

cut -d ' ' -f1 "$TMPDIR/line" | tr -d '[:space:]' \
> "$TMPDIR/hash_online"

openssl dgst -sha512 putty.zip | cut -d '=' -f2 | tr -d '[:space:]' \
> "$TMPDIR/hash_local"

if cmp -s "$TMPDIR/hash_online" "$TMPDIR/hash_local"
then
    echo "HASH OK"
else
    echo "HASH NOT OK"
fi

```

script

```
#!/bin/bash

TMPDIR=$(mktemp -d)

# checksum
curl -L -o "$TMPDIR/sha512sums" \
https://the.earth.li/~sgtatham/putty/latest/sha512sums


#signatur
curl -L -o "$TMPDIR/sha512sums.gpg" \
https://the.earth.li/~sgtatham/putty/latest/sha512sums.gpg


gpg --verify "$TMPDIR/sha512sums.gpg" 

if [ $? -ne 0 ]
then
    echo "SIGNATURE INVALID"
    exit 1
fi

echo "SIGNATURE OK"



grep "w64/putty.zip$" "$TMPDIR/sha512sums" > "$TMPDIR/line"

cut -d ' ' -f1 "$TMPDIR/line" | tr -d '[:space:]' \
> "$TMPDIR/hash_online"

openssl dgst -sha512 putty.zip | cut -d '=' -f2 | tr -d '[:space:]' \
> "$TMPDIR/hash_local"

if cmp -s "$TMPDIR/hash_online" "$TMPDIR/hash_local"
then
    echo "HASH OK"
else
    echo "HASH NOT OK"
fi
```
